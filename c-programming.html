Computer Basic C Programming
Format Specifier Data Type Description / Use Case
%d or %i int Signed decimal integer.
%c char Single character.
%f float Floating-point number (decimal).
%lf double Double-precision floating-point number.
%s char[] or char* String of characters.
%u unsigned int Unsigned decimal integer (positive numbers only).
%ld long int Long signed integer.
%x or %X unsigned int Unsigned hexadecimal integer (%x for lowercase, %X for uppercase).
%o unsigned int Unsigned octal integer.
%p Pointer Memory address (prints in hexadecimal format).
Control Flow Syntax

1. if / else if / else

if (condition) {
    // Executes if condition is true
} else if (another_condition) {
    // Executes if the above is false and this is true
} else {
    // Executes if all conditions are false
}

2. switch (with break)

switch (variable) {
    case 1:
        // Executes if variable equals 1
        break; // Exits the switch block to prevent fall-through
    case 2:
        // Executes if variable equals 2
        break;
    default:
        // Executes if no cases match
}

3. for loop (with continue)

for (int i = 0; i < 10; i++) {
    if (i == 5) {
        continue; // Skips the rest of the code for i=5 and jumps to i=6
    }
    // Code to repeat
}

4. while loop

while (condition) {
    // Code repeats as long as condition evaluates to true
    // (Ensure you update the condition variable inside to avoid infinite loops)
}

5. do...while loop

do {
    // Code executes AT LEAST ONCE, then repeats if condition is true
} while (condition); // The semicolon here is mandatory!
